

 bool bfs(vector<vector<int>>& adj,vector<int> &color,int node){
        color[node]=0;
        
        queue<int> q;
        q.push(node);
        
        while(q.empty()==false){
            int temp=q.front();
            q.pop();
            
            for(auto ele:adj[temp]){
                if(color[ele]==-1){
                    color[ele]=color[temp]==0 ? 1:0;
                    q.push(ele);
                    
                }
                else if(color[ele]==color[temp]) {
                    return false;
                    
                }
            }
        }
        return true;
    }

    bool isvalid(vector<vector<int>>& adj,int node,vector<int> & color,int curr) {
    color[node]=curr;

    for(auto ele:adj[node]){
        if(color[ele]==-1){
            if(isvalid(adj,ele,color,!color[node])==false)
                return false;
        }
        else if(color[ele]==color[node])
            return false;
    }

    return true;
    }
